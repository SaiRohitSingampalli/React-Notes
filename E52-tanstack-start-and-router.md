# E52 — A TanStack Start app

> Runnable code for [52 — TanStack Start and TanStack Router](52-tanstack-start-and-router.md). 3 tests.

The chapter's app: one router per request, validated search params as state, server functions with an auth middleware, a server route for webhooks, a pathless `_authed` layout whose `beforeLoad` gates the UI, global CSRF middleware in `start.ts`, and the experimental deferred hydration on the home page. The services are stand-ins. `serve.mjs` is a minimal Node runner for the built fetch handler.

`npm test` builds the app, starts it, and checks the guard: no session redirects to `/login?redirect=/invoices`, and a session cookie lets the route render. `src/routeTree.gen.ts` is generated by the router plugin; it's included so the type-check works before a first build, and the build regenerates it.

```ts
// src/functions/invoices.functions.ts: server functions are RPC endpoints; each one validates and authorises
import { createServerFn } from '@tanstack/react-start';
import { notFound } from '@tanstack/react-router';
import { z } from 'zod';
import { authMiddleware } from './auth';
import { invoiceRepo } from '~/server/invoices.server';
import { readSessionUser } from '~/server/auth.server';

export const getSessionUser = createServerFn({ method: 'GET' }).handler(async () => {
  const user = await readSessionUser();
  return user ? { id: user.id, name: user.name } : null; // DTO: the client never sees permissions or tenant
});

export const listInvoices = createServerFn({ method: 'GET' })
  .middleware([authMiddleware])
  .validator(z.object({
    page: z.number().int().min(1),
    status: z.enum(['open', 'paid', 'void']).optional(),
    sort: z.enum(['number', '-number']),
  }))
  .handler(async ({ data, context }) => invoiceRepo.list(context.user.tenantId, data));

export const getInvoice = createServerFn({ method: 'GET' })
  .middleware([authMiddleware])
  .validator(z.object({ id: z.string().min(1) }))
  .handler(async ({ data, context }) => {
    const invoice = await invoiceRepo.get(context.user.tenantId, data.id);
    if (!invoice) throw notFound();
    return { invoice, canVoid: context.user.permissions.includes('invoice:void') && invoice.status === 'open' };
  });

export const voidInvoice = createServerFn({ method: 'POST' })
  .middleware([authMiddleware])
  .validator(z.object({ id: z.string().min(1), reason: z.string().trim().min(5) }))
  .handler(async ({ data, context }) => {
    if (!context.user.permissions.includes('invoice:void')) throw new Error('Forbidden');
    await invoiceRepo.void(context.user.tenantId, data.id, data.reason);
    return { ok: true as const };
  });
```

### `src/router.tsx`

```tsx
// src/router.tsx: one router per request on the server, one per tab in the browser
import { createRouter } from '@tanstack/react-router';
import { QueryClient } from '@tanstack/react-query';
import { setupRouterSsrQueryIntegration } from '@tanstack/react-router-ssr-query';
import { routeTree } from './routeTree.gen';

export function getRouter() {
  const queryClient = new QueryClient({ defaultOptions: { queries: { staleTime: 30_000 } } });
  const router = createRouter({
    routeTree,
    context: { queryClient },
    defaultPreload: 'intent',          // preload a route's code and data on hover/focus
    defaultPreloadStaleTime: 0,        // let TanStack Query decide freshness, not the router cache
    scrollRestoration: true,
  });
  setupRouterSsrQueryIntegration({ router, queryClient }); // dehydrate on the server, hydrate in the browser
  return router;
}

declare module '@tanstack/react-router' {
  interface Register { router: ReturnType<typeof getRouter> } // makes every Link, navigate and hook type-safe
}
```

### `src/routes/_authed/invoices/$invoiceId.tsx`

```tsx
// src/routes/_authed/invoices/$invoiceId.tsx: mutation, then invalidation (the chapter prints an excerpt)
import { useState } from 'react';
import { createFileRoute, useRouter, type ErrorComponentProps } from '@tanstack/react-router';
import { useServerFn } from '@tanstack/react-start';
import { getInvoice, voidInvoice } from '~/functions/invoices.functions';

export const Route = createFileRoute('/_authed/invoices/$invoiceId')({
  loader: ({ params }) => getInvoice({ data: { id: params.invoiceId } }),  // params.invoiceId: string
  head: ({ loaderData }) => ({ meta: [{ title: loaderData ? `Invoice ${loaderData.invoice.number}` : 'Invoice' }] }),
  pendingComponent: () => <p>Loading invoice…</p>,
  errorComponent: InvoiceError,
  notFoundComponent: () => <p>That invoice doesn't exist.</p>,
  component: InvoiceDetail,
});

function InvoiceDetail() {
  const { invoice, canVoid } = Route.useLoaderData();
  const router = useRouter();
  const voidFn = useServerFn(voidInvoice);          // handles redirect() and notFound() thrown by the server
  const [error, setError] = useState<string | null>(null);
  const [pending, setPending] = useState(false);

  async function onSubmit(form: FormData) {
    setPending(true); setError(null);
    try {
      await voidFn({ data: { id: invoice.id, reason: String(form.get('reason') ?? '') } });
      await router.invalidate({ sync: true });          // re-run active loaders, then render the persisted state
    } catch (e) {
      setError(e instanceof Error ? e.message : 'Could not void the invoice');
    } finally { setPending(false); }
  }

  return (
    <article>
      <h1>Invoice {invoice.number}</h1>
      <p>Status: {invoice.status}</p>
      {canVoid && (
        <form action={onSubmit}>
          <label htmlFor="reason">Reason for voiding</label>
          <input id="reason" name="reason" aria-invalid={!!error} aria-describedby={error ? 'void-error' : undefined} />
          {error && <p id="void-error" role="alert">{error}</p>}
          <button type="submit" aria-disabled={pending}>{pending ? 'Voiding…' : 'Void invoice'}</button>
        </form>
      )}
    </article>
  );
}

function InvoiceError({ reset }: ErrorComponentProps) {
  // Don't render raw error messages; they're reported through the router's onError hook instead
  return <div role="alert"><p>Couldn't load this invoice.</p><button onClick={reset}>Try again</button></div>;
}
```

```tsx
// src/routes/_authed/invoices/index.tsx: the URL is the state; search params are validated and typed
import { createFileRoute, Link, stripSearchParams } from '@tanstack/react-router';
import { queryOptions, useSuspenseQuery } from '@tanstack/react-query';
import { z } from 'zod';
import { listInvoices } from '~/functions/invoices.functions';

const invoiceSearch = z.object({
  page: z.number().int().min(1).catch(1),                         // bad input degrades to a default
  status: z.enum(['open', 'paid', 'void']).optional().catch(undefined),
  sort: z.enum(['number', '-number']).catch('number'),
});
type InvoiceSearch = z.infer<typeof invoiceSearch>;

const invoicesQuery = (s: InvoiceSearch) =>
  queryOptions({ queryKey: ['invoices', s], queryFn: () => listInvoices({ data: s }) });

export const Route = createFileRoute('/_authed/invoices/')({
  validateSearch: invoiceSearch,
  search: { middlewares: [stripSearchParams({ page: 1, sort: 'number' })] }, // keep default values out of URLs
  loaderDeps: ({ search }) => search,                          // the loader's cache key: only what it reads
  loader: ({ context, deps }) => context.queryClient.ensureQueryData(invoicesQuery(deps)),
  component: Invoices,
});

function Invoices() {
  const search = Route.useSearch();
  const { data } = useSuspenseQuery(invoicesQuery(search));
  const navigate = Route.useNavigate();
  return (
    <section>
      <h1>Invoices ({data.total})</h1>
      <label>
        Status{' '}
        <select
          value={search.status ?? ''}
          onChange={(e) => navigate({
            search: (prev) => ({ ...prev, page: 1, status: e.target.value ? (e.target.value as InvoiceSearch['status']) : undefined }),
          })}
        >
          <option value="">All</option><option value="open">Open</option><option value="paid">Paid</option><option value="void">Void</option>
        </select>
      </label>
      <ul>
        {data.items.map((inv) => (
          <li key={inv.id}>
            <Link to="/invoices/$invoiceId" params={{ invoiceId: inv.id }}>{inv.number}</Link> · {inv.status}
          </li>
        ))}
      </ul>
      <Link from={Route.fullPath} search={(prev) => ({ ...prev, page: prev.page + 1 })} disabled={search.page * 20 >= data.total}>
        Next page
      </Link>
    </section>
  );
}
```

### `src/server/invoices.server.ts`

```ts
export type InvoiceStatus = 'open' | 'paid' | 'void';
export type InvoiceDTO = { id: string; number: string; status: InvoiceStatus; totalMinor: number; currency: string };
const rows: InvoiceDTO[] = Array.from({ length: 45 }, (_, i) => ({
  id: `inv_${i + 1}`, number: `INV-${String(i + 1).padStart(4, '0')}`,
  status: (['open', 'paid', 'void'] as const)[i % 3], totalMinor: 10_000 + i * 731, currency: 'EUR',
}));
export const invoiceRepo = {
  async list(_tenantId: string, q: { page: number; status?: InvoiceStatus; sort: 'number' | '-number' }) {
    let r = q.status ? rows.filter((x) => x.status === q.status) : rows;
    r = q.sort === '-number' ? [...r].reverse() : r;
    return { total: r.length, items: r.slice((q.page - 1) * 20, q.page * 20) };
  },
  async get(_tenantId: string, id: string) { return rows.find((x) => x.id === id) ?? null; },
  async void(_tenantId: string, id: string, _reason: string) {
    const r = rows.find((x) => x.id === id); if (r) r.status = 'void';
  },
};
```

### `serve.mjs`

```js
// minimal Node runner for the Start fetch handler (static assets + SSR)
import { serve } from 'srvx';
import { readFile } from 'node:fs/promises';
import { extname, join } from 'node:path';
import server from './dist/server/server.js';
const types = { '.js': 'text/javascript', '.css': 'text/css' };
serve({
  port: Number(process.env.PORT ?? 3952),
  async fetch(req) {
    const { pathname } = new URL(req.url);
    if (pathname.startsWith('/assets/')) {
      try { return new Response(await readFile(join('dist/client', pathname)), { headers: { 'content-type': types[extname(pathname)] ?? 'application/octet-stream' } }); } catch {}
    }
    return server.fetch(req);
  },
});
```
